cmake_minimum_required(VERSION 3.18)
project(aerobench_sim C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# --------------------------------------------------------------------
# Toolchain selection
# Build with:
#   cmake -B build/arm   -DTARGET_ARCH=arm   (runs inside QEMU)
#   cmake -B build/host  -DTARGET_ARCH=host  (runs on WSL2 for dev)
# --------------------------------------------------------------------

option(TARGET_ARCH "Build target: arm | host" "host")

if(TARGET_ARCH STREQUAL "arm")
    set(CMAKE_C_COMPILER    arm-linux-gnueabihf-gcc)
    set(CMAKE_LINKER        arm-linux-gnueabihf-ld)
    add_compile_options(-march=armv7-a -mfpu=vfpv3 -mfloat-abi=hard)
    message(STATUS "Cross-compiling for ARM (QEMU target)")
else()
    message(STATUS "Building for host (development/test)")
endif()

add_compile_options(
    -Wall -Wextra -Wpedantic
    -O2
    -g
)

# --------------------------------------------------------------------
# Sources
# --------------------------------------------------------------------

set(SIM_SOURCES
    src/main_sim.c
    src/scenario_engine.c
    src/sensor_imu.c
    src/sensor_gps.c
    src/sensor_battery.c
    src/sensor_baro.c
    src/fault_injector.c
    src/packet_serialiser.c
    src/uart_bridge.c
    src/tcp_server.c
)

include_directories(include)

add_executable(aerobench-sim ${SIM_SOURCES})

target_link_libraries(aerobench-sim m pthread)

# --------------------------------------------------------------------
# Install scenarios alongside binary
# --------------------------------------------------------------------

install(TARGETS aerobench-sim DESTINATION bin)
install(DIRECTORY scenarios/ DESTINATION share/aerobench/scenarios)
